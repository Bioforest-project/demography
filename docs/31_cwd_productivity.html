Composition change data

Formula for the contribution of fluxes

The contribution of a given flux (recruitment, growth or mortality) at time \(t\) can be written as:

\[CF_{F,t} = \sum_{i\in F_t}\frac{ba_{i,t} (ft_{i} - CWM_{t0})}{BA_{t}}-\sum_{i\in F_t}\frac{ba_{i,t0} (ft_{i} - CWM_{t0})}{BA_{t0}}\]

where \(F_t\) is the set of trees contributing to a given flux between \(t-1\) and \(t\) (recruitment: trees recruited at \(t\); growth: trees that were measured and alive at \(t\) and \(t-1\); mortality: trees that died between \(t-1\) and \(t\)).

\(ba_{i,t}\) is the basal area of a tree \(i\) at time \(t\); \(ft_{i}\) is the functional trait value of tree \(i\); \(BA_t\) is the basal area of the entire community at \(t\); \(CWM_t\) is the community weighted mean trait value at \(t\).

We subtract \(CWM_{t-1}\) to have negative contributions when the flux decreases the CWM between \(t-1\) and \(t\); and positive contributions that increase the \(CWM\).

As written, the contributions are additive:

\[CWM_t= CWM_{t-1} + CF_{G,t} + CF_{R,t}+CF_{M,t}\]

plot_info <- read.csv("data/raw_data/bioforest-plot-information.csv") |>
  select(site, plot, Year_of_harvest, Treatment)
data_ba <- read.csv("data/derived_data/aggregated_data_v10.csv") |>
  filter(grepl("badiff", variable)) |>
  separate(variable, c(NA, "cohort", "flux"), sep = "_") |>
  rename(diffba = value)
data_ba_all <- data_ba |>
  group_by(Site, Plot, Year) |>
  summarise(
    diffba = sum(diffba), cohort = "all",
    flux = "all", .groups = "drop"
  )
data <- read.csv("data/derived_data/aggregated_data_v10.csv") |>
  filter(grepl("cwmdiff", variable)) |>
  left_join(plot_info, by = join_by(Site == site, Plot == plot)) |>
  separate(variable, c("trait", NA, "cohort", "flux"), sep = "_") |>
  mutate(flux = ifelse(is.na(flux), "all", flux)) |>
  left_join(bind_rows(data_ba, data_ba_all)) |>
  mutate(t0 = case_when(
    Site == "Paracou" & cohort == "s" ~ 1990,
    Site == "Paracou" & cohort != "s" ~ 1987,
    Site == "Mbaiki" & cohort == "s" ~ 1989,
    Site == "Mbaiki" & cohort != "s" ~ 1986,
    Site == "SUAS" ~ 1993
  )) |>
  filter(Year >= t0)

Contribution to CWM change vs BA change

Code 
sites <- c("Paracou", "Mbaiki", "SUAS")
data_cwm0 <- read.csv("data/derived_data/aggregated_data_v10.csv") |>
  filter(grepl("_cwm_ba$|^ba$", variable) & Site %in% sites) |>
  pivot_wider(names_from = "variable") |>
  pivot_longer(cols = contains("cwm")) |>
  mutate(trait = gsub("_cwm_ba$", "", name)) |>
  select(-name) |>
  mutate(Plot = gsub("_[0-9]+", "", Plot)) |>
  group_by(Site, Plot, trait, Year) |>
  summarise(cwm = weighted.mean(value, ba), ba = mean(ba)) |>
  group_by(Site, Plot, trait) |>
  summarise(cwm = cwm[Year == min(Year)], ba = ba[Year == min(Year)])
gmain <- data |>
  filter(cohort != "all") |>
  mutate(group = paste(cohort, flux, sep = "_")) |>
  left_join(data.frame(lab = labs_flux, group = names(labs_flux))) |>
  left_join(data_cwm0) |>
  group_by(Site, Treatment, Plot, trait, group = lab, Year) |>
  summarise(value = sum(value / cwm) * 100, diffba = sum(diffba / ba) * 100) |>
  group_by(Site, Plot, trait, group) |>
  arrange(Year) |>
  mutate(valuec = cumsum(value), diffbac = cumsum(diffba)) |>
  ggplot(aes(
    x = diffbac, y = valuec,
    group = paste(Plot, group), col = paste(group, Treatment)
  )) +
  geom_vline(xintercept = 0, lty = 2, col = "grey") +
  geom_hline(yintercept = 0, lty = 2, col = "grey") +
  geom_path() +
  geom_point() +
  facet_grid(trait ~ Site, scales = "free_y") +
  scale_color_manual(values = RColorBrewer::brewer.pal(10, "Paired")
  [c(1:4, 9:10, 7:8, 5:6)]) +
  labs(
    x = "Cumulative change in BA (% of initial BA)", col = NULL,
    y = "Cumulative contribution to CWM change (% of initial CWM)"
  ) +
  theme_minimal() +
  theme(legend.position = "none", panel.grid = element_blank())
